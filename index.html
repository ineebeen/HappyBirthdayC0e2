            <h1>🎉 생일 축하해!</h1>

            <p>소중한 친구에게,</p><br>

            <p>여기에 준비한 생일 편지 텍스트를 작성하시면 됩니다.</p><br>

            <p>내용이 길어지면 모바일 화면에서 손가락으로 위아래 스크롤을 하면서 읽을 수 있습니다.</p><br>

            <p>오늘 하루 행복하고 즐거운 일만 가득하길 바랄게!</p>

    <script>

        const envelope = document.getElementById('envelope');

        const foldedLetter = document.getElementById('foldedLetter');

        const fullLetter = document.getElementById('fullLetter');

        const closeBtn = document.getElementById('closeBtn');


        let startX = 0;

        let startY = 0;

        let isOpen = false;


        // 터치 이벤트 (모바일 스와이프 감지)

        envelope.addEventListener('touchstart', (e) => {

            startX = e.touches[0].clientX;

            startY = e.touches[0].clientY;

        });


        envelope.addEventListener('touchend', (e) => {

            const endX = e.changedTouches[0].clientX;

            const endY = e.changedTouches[0].clientY;


            const diffX = endX - startX;

            const diffY = endY - startY;


            // 좌/우 스와이프 (100px 이상 움직였을 때) -> 봉투 뒷면 전환

            if (Math.abs(diffX) > 100 && Math.abs(diffY) < 50 && !isOpen) {

                envelope.classList.toggle('flipped');

            }

            

            // 위로 스와이프 (위쪽으로 80px 이상) -> 봉투 열림 및 편지지 올라옴

            if (diffY < -80 && Math.abs(diffX) < 50 && !isOpen) {

                envelope.classList.remove('flipped');

                envelope.classList.add('opened');

                isOpen = true;

            }

        });


        // 위로 올라온 접힌 편지지 터치 시 전체화면 확장

        foldedLetter.addEventListener('click', (e) => {

            if (isOpen) {

                e.stopPropagation();

                fullLetter.classList.add('active');

            }

        });


        // X 버튼 클릭 시 초기 상태로 복구

        closeBtn.addEventListener('click', () => {

            fullLetter.classList.remove('active');

            envelope.classList.remove('opened');

            isOpen = false;

        });

    </script>
